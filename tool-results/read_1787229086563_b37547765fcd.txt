   138→function groupForMethod(method: PaymentMethod): MethodGroup {
   139→  if (method === "CASH") return "CASH"
   140→  if (method === "BANK_TRANSFER" || method === "CHEQUE") return "BANK"
   141→  return "MOBILE"
   142→}
   143→
   144→function fmtDate(iso: string | null): string {
   145→  if (!iso) return "—"
   146→  return new Date(iso).toLocaleDateString(undefined, {
   147→    day: "2-digit",
   148→    month: "short",
   149→    year: "numeric",
   150→  })
   151→}
   152→
   153→function fmtDateTime(iso: string | null): string {
   154→  if (!iso) return "—"
   155→  return new Date(iso).toLocaleString(undefined, {
   156→    day: "2-digit",
   157→    month: "short",
   158→    year: "numeric",
   159→    hour: "2-digit",
   160→    minute: "2-digit",
   161→  })
   162→}

   172→function StatusBadge({ status }: { status: string }) {
   173→  const s = STATUS_STYLES[status] ?? STATUS_STYLES.PENDING

   178→    </Badge>
   179→  )

   183→export default function DepositRequestClient({
   184→  memberId,
   185→  member,
   186→  collectionTypes,
   187→  bankAccounts,
   188→  missingGroups,
   189→  recentRequests,
   190→}: Props) {
   191→  const router = useRouter()
   192→  const fileInputRef = useRef<HTMLInputElement>(null)
   193→  const editFileInputRef = useRef<HTMLInputElement>(null)
   194→  const [isPending, startTransition] = useTransition()
   195→
   196→  // Form fields (initial submission)
   197→  const [amount, setAmount] = useState("")
   198→  const [collectionTypeId, setCollectionTypeId] = useState("")
   199→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>("BKASH")
   200→  const [transactionDate, setTransactionDate] = useState(new Date().toISOString().slice(0, 10))
   201→  const [referenceNo, setReferenceNo] = useState("")
   202→  const [notes, setNotes] = useState("")
   203→  const [slipFile, setSlipFile] = useState<File | null>(null)
   204→
   205→  // ── Resubmit dialog state ("Returned for correction" path) ──────────────
   206→  // When a member clicks "Edit & Resubmit" on a RETURNED request, we open a
   207→  // dialog pre-filled with the original values + the admin's return reason.
   208→  // The member can edit any field, optionally attach a NEW slip (otherwise
   209→  // the original is kept), and submit. Calls `resubmitDepositRequest`.
   210→  const [editOpen, setEditOpen] = useState(false)
   211→  const [editingId, setEditingId] = useState<string | null>(null)
   212→  const [editAmount, setEditAmount] = useState("")
   213→  const [editCollectionTypeId, setEditCollectionTypeId] = useState("")
   214→  const [editPaymentMethod, setEditPaymentMethod] = useState<PaymentMethod>("BKASH")
   215→  const [editTransactionDate, setEditTransactionDate] = useState("")
   216→  const [editReferenceNo, setEditReferenceNo] = useState("")
   217→  const [editNotes, setEditNotes] = useState("")
   218→  const [editSlipFile, setEditSlipFile] = useState<File | null>(null)
   219→  const [editReturnReason, setEditReturnReason] = useState("")
   220→  const [editVoucherNo, setEditVoucherNo] = useState<string | null>(null)
   221→  const [editExistingSlipUrl, setEditExistingSlipUrl] = useState<string | null>(null)
   222→  const [editExistingSlipName, setEditExistingSlipName] = useState<string | null>(null)
   223→

   225→  // the member knows where to send money.

   227→    CASH: [],
   228→    BANK: [],
   229→    MOBILE: [],
   230→  }

   237→  const groupMissing = missingGroups.includes(groupForMethod(paymentMethod))
   238→  const editActiveMethodGroup = METHOD_GROUPS.find(
   239→    (g) => g.group === groupForMethod(editPaymentMethod)
   240→  )
   241→  const editGroupMissing = missingGroups.includes(groupForMethod(editPaymentMethod))
   242→

   244→    const f = e.target.files?.[0] ?? null
   245→    setSlipFile(f)
   246→  }

   249→    e.preventDefault()

   251→    if (!amt || amt <= 0) {
   252→      toast.error("Invalid amount", { description: "Please enter a valid deposit amount." })
   253→      return
   254→    }
   255→    if (!slipFile) {
   256→      toast.error("Deposit slip required", { description: "Please attach the deposit slip / transaction document as proof." })
   257→      return
   258→    }
   259→    if (slipFile.size > 10 * 1024 * 1024) {
   260→      toast.error("File too large", { description: "Maximum slip file size is 10 MB." })
   261→      return
   262→    }
   263→
   264→    const formData = new FormData()
   265→    formData.append("amount", String(amt))
   266→    formData.append("method", paymentMethod)
   267→    if (collectionTypeId) formData.append("collectionTypeId", collectionTypeId)
   268→    if (referenceNo.trim()) formData.append("referenceNo", referenceNo.trim())
   269→    if (transactionDate) formData.append("transactionDate", transactionDate)
   270→    if (notes.trim()) formData.append("notes", notes.trim())
   271→    formData.append("slip", slipFile)
   272→
   273→    startTransition(async () => {
   274→      const res = await submitDepositRequest(memberId, formData)
   275→      if (res.ok) {

   277→          description: "Your request is pending admin approval. You'll be notified once it's reviewed.",
   278→        })

   281→        setReferenceNo("")
   282→        setNotes("")

   284→        setCollectionTypeId("")
   285→        if (fileInputRef.current) fileInputRef.current.value = ""

   288→        toast.error("Submission failed", { description: res.error })
   289→      }
   290→    })
   291→  }
   292→

   296→    setEditAmount(r.amount ? String(r.amount) : "")

   298→    setEditPaymentMethod((r.method as PaymentMethod) ?? "BKASH")
   299→    setEditTransactionDate(
   300→      r.transactionDate ? r.transactionDate.slice(0, 10) : new Date().toISOString().slice(0, 10)
   301→    )
   302→    setEditReferenceNo(r.referenceNo ?? "")
   303→    setEditNotes(r.notes ?? "")
   304→    setEditSlipFile(null)
   305→    setEditReturnReason(r.returnReason ?? "")
   306→    setEditVoucherNo(r.voucherNo)
   307→    const slip = r.attachments?.[0]
   308→    setEditExistingSlipUrl(slip?.url ?? null)
   309→    setEditExistingSlipName(slip?.name ?? null)
   310→    if (editFileInputRef.current) editFileInputRef.current.value = ""
   311→    setEditOpen(true)
   312→  }
   313→
   314→  const handleEditFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   315→    const f = e.target.files?.[0] ?? null
   316→    setEditSlipFile(f)
   317→  }
   318→
   319→  const handleEditSubmit = (e: React.FormEvent<HTMLFormElement>) => {
   320→    e.preventDefault()
   321→    if (!editingId) return
   322→    const amt = parseFloat(editAmount)
   323→    if (!amt || amt <= 0) {
   324→      toast.error("Invalid amount", { description: "Please enter a valid deposit amount." })
   325→      return
   326→    }
   327→    if (editSlipFile && editSlipFile.size > 10 * 1024 * 1024) {
   328→      toast.error("File too large", { description: "Maximum slip file size is 10 MB." })
   329→      return
   330→    }
   331→
   332→    const formData = new FormData()
   333→    formData.append("amount", String(amt))
   334→    formData.append("method", editPaymentMethod)
   335→    if (editCollectionTypeId) formData.append("collectionTypeId", editCollectionTypeId)
   336→    if (editReferenceNo.trim()) formData.append("referenceNo", editReferenceNo.trim())
   337→    if (editTransactionDate) formData.append("transactionDate", editTransactionDate)
   338→    if (editNotes.trim()) formData.append("notes", editNotes.trim())
   339→    if (editSlipFile) {
   340→      formData.append("slip", editSlipFile)
   341→    } else {
   342→      // Send an empty File so the action sees `slipFile` as null and keeps
   343→      // the original. FormData.append with a Blob is required for the field
   344→      // to exist; we send an empty 0-byte Blob named "slip" so the action's
   345→      // `slipFile.size === 0` branch fires correctly.
   346→      formData.append("slip", new Blob([], { type: "application/octet-stream" }), "")
   347→    }
   348→
   349→    startTransition(async () => {
   350→      const res = await resubmitDepositRequest(memberId, editingId, formData)
   351→      if (res.ok) {
   352→        toast.success("Deposit request resubmitted", {
   353→          description: "Your updated request is back in the admin approval queue.",
   354→        })
   355→        setEditOpen(false)
   356→        setEditingId(null)
   357→        router.refresh()
   358→      } else {
   359→        toast.error("Resubmission failed", { description: res.error })
   360→      }
   361→    })
   362→  }
   363→
   364→  return (
   365→    <div className="space-y-6">
   366→      {/* Header */}
   367→      <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-4">
   368→        <div>
   369→          <h1 className="text-2xl sm:text-3xl font-bold tracking-tight text-slate-900 dark:text-white flex items-center gap-2">
   370→            <ArrowDownToLine className="h-7 w-7 text-emerald-500" /> Deposit Request
   371→          </h1>
   372→          <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">
   373→            Already deposited to the Somiti account? Submit the details with your deposit slip — admin will verify and credit your balance.

   375→        </div>
   376→        <div className="flex items-center gap-3 rounded-2xl border border-emerald-200 dark:border-emerald-900/40 bg-emerald-50 dark:bg-emerald-950/30 px-4 py-2">
   377→          <Wallet className="h-5 w-5 text-emerald-600" />
   378→          <div>
   379→            <p className="text-[10px] uppercase tracking-widest font-bold text-slate-500 dark:text-slate-400">
   380→              Current Balance
   381→            </p>
   382→            <p className="text-lg font-extrabold text-emerald-600 dark:text-emerald-400">
   383→              ৳ {member.currentBalance.toLocaleString()}
   384→            </p>
   385→          </div>
   386→        </div>
   387→      </div>
   388→
   389→      {/* Where to deposit (read-only bank accounts) */}
   390→      <Card className="bg-white dark:bg-slate-900 border border-slate-200/70 dark:border-slate-800/70 shadow-sm rounded-2xl overflow-hidden">
   391→        <CardHeader className="bg-slate-50/80 dark:bg-slate-900/80 border-b border-slate-200/60 dark:border-slate-800/60 px-6 py-4">
   392→          <CardTitle className="flex items-center gap-2 text-sm text-slate-800 dark:text-white">
   393→            <Building2 className="h-4 w-4 text-indigo-500" /> Somiti Accounts — Deposit Here First
   394→          </CardTitle>
   395→        </CardHeader>
   396→        <CardContent className="p-6">
   397→          {bankAccounts.length === 0 ? (
   398→            <div className="flex items-start gap-2 rounded-xl border border-amber-200 dark:border-amber-900/50 bg-amber-50 dark:bg-amber-950/20 p-3 text-sm text-amber-700 dark:text-amber-400">
   399→              <AlertCircle className="h-4 w-4 mt-0.5 shrink-0" />
   400→              <div>
   401→                No active bank/mobile accounts configured. Please contact management for deposit instructions before submitting a request.
   402→              </div>
   403→            </div>
   404→          ) : (
   405→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">

   407→                const groupCfg = METHOD_GROUPS.find((m) => m.group === g)!
   408→                const accounts = banksByGroup[g]
   409→                if (accounts.length === 0) return null
   410→                const Icon = groupCfg.icon

   412→                  <div
   413→                    key={g}
   414→                    className="rounded-xl border border-slate-200 dark:border-slate-800 bg-slate-50/50 dark:bg-slate-950/50 p-4"
   415→                  >
   416→                    <div className="flex items-center gap-2 mb-3">
   417→                      <div className="h-8 w-8 rounded-lg bg-indigo-100 dark:bg-indigo-950/40 flex items-center justify-center">

   449→                            {PAYMENT_METHOD_LABEL[a.paymentMethod]}
   450→                          </p>
   451→                        </li>
   452→                      ))}
   453→                    </ul>
   454→                  </div>
   455→                )
   456→              })}
   457→            </div>
   458→          )}
   459→        </CardContent>
   460→      </Card>
   461→
   462→      {/* Submission form + recent requests */}
   463→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-6">
   464→        {/* Form (3 cols) */}
   465→        <div className="lg:col-span-3">
   466→          <Card className="bg-white dark:bg-slate-900 border border-slate-200/70 dark:border-slate-800/70 shadow-sm rounded-2xl overflow-hidden">
   467→            <CardHeader className="bg-slate-50/80 dark:bg-slate-900/80 border-b border-slate-200/60 dark:border-slate-800/60 px-6 py-4">
   468→              <CardTitle className="flex items-center gap-2 text-sm text-slate-800 dark:text-white">
   469→                <FileText className="h-4 w-4 text-emerald-500" /> Submit New Deposit Request
